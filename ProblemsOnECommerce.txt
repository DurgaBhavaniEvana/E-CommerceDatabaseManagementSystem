1.Retrieve all products along with their total sales revenue from completed orders.

select p.product_name,sum(s.quantity*s.price_per_unit) Total_sales
from products p inner join sales s
on p.product_id=s.product_id
where s.order_status='Completed'
group by p.product_id,p.product_name;

2.List all customers and the products they have purchased, showing only those who have ordered more than two products.

select c.customer_name,p.product_name
from products p inner join sales s
on p.product_id=s.product_id
join customers c
on c.customer_id=s.customer_id
where c.customer_id in (select c.customer_id
                      from sales s inner join customers c
                      on c.customer_id=s.customer_id
                      group by c.customer_id
                      having count(distinct s.product_id)>2);

3.Find the average order value per customer for orders with a quantity of more than 5.

select c.customer_name,round(avg(s.quantity*s.price_per_unit),2) avg_order_vale
from customers c inner join sales s
on c.customer_id=s.customer_id
where s.quantity >5
group by c.customer_id,c.customer_name;

4.Get the top 5 customers by total spending on 'Accessories'.

select customer_name,total
from(select c.customer_name,sum(s.quantity*s.price_per_unit) total,dense_rank() over(order by sum(s.quantity*s.price_per_unit) desc) d_rank
from customers c inner join sales s
on c.customer_id=s.customer_id
                       inner join products p
on s.product_id=p.product_id
where p.category='Accessories'
group by c.customer_name,c.customer_id)
where d_rank<=5;

5.Find the most popular product based on total quantity sold in 2023.

select product_name
from(select p.product_name,sum(quantity) sold_items,dense_rank() over(order by sum(quantity) desc) rank_s
from sales s inner join products p
on s.product_id=p.product_id
where order_status='Completed' and extract(year from order_date) = 2023
group by p.product_id,p.product_name)
where rank_s=1;

7.Retrieve the total quantity of products sold by category in 2023.

8.Find the customers who have made the most purchases in a single month.

select customer_id,customer_name
from(select c.customer_id,c.customer_name,count(*),dense_rank() over(order by count(*) desc) rank1
from customers c inner join sales s
on c.customer_id=s.customer_id
group by c.customer_id,c.customer_name,to_char(s.order_date,'mon'))
where rank1=1;

21.Get the average payment amount per customer who has placed more than 3 orders (use INNER JOIN between payments and sales, group by customer, and apply a HAVING clause).

select customer_id,round(avg(sum_amount),2)
from(select c.customer_id,sum(quantity*price_per_unit) sum_amount
from sales s inner join customers c
on s.customer_id=c.customer_id
                    join payments p
on s.order_id=p.order_id
where p.payment_status='Payment Successed'
group by c.customer_id
having count(*)>3)
group by customer_id;

22.Retrieve the total sales for each product category and only include categories where the total sales exceed 100,000 (use INNER JOIN between sales and products, group by category).

23.Show the number of customers in each state who have made purchases with a total spend greater than 50,000 (use INNER JOIN between sales and customers).

select c.state,count(*) no_of_customers
from sales s inner join customers c
on s.customer_id=c.customer_id
where c.customer_id in (select customer_id
                      from sales
                      group by customer_id
                      having sum(quantity*price_per_unit)>50000)
group by c.state;

28.Get the number of orders per product and filter to include only products that have been ordered more than 10 times, then order the results by the highest number of orders (use INNER JOIN between sales and products).

select p.product_id,p.product_name,count(*) no_of_orders
from sales s inner products p
on s.product_id=p.product_id
group by p.product_id,p.product_name
having count(*)>10
order by no_of_orders desc;

29.Retrieve the number of payments made per customer where the payment status is 'Payment Successed' and group by customer, ordering by payment count (use INNER JOIN between payments and customers).

select c.customer_id,c.customer_name,count(*) no_of_payments 
from sales s inner join customers c
on s.customer_id=c.customer_id
                 inner join payments p
on s.order_id=p.order_id
where p.payment_status='Payment Successed'
group by c.customer_id,c.customer_name
order by no_of_payments;
